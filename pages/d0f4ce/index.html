Java 使用poi插入附件注意事项

# excel表格插入附件

  1. 注意addOlePackage里的文件名如果有中文会出现乱码导致文件名错误打不开文件
  2. 如果read字节流是空的[],那么插入的附件即使能够打开也是会显示文档乱码需要修复
// 导出清单的附件
ResourceLoader resourceLoader = new DefaultResourceLoader();
Resource resource = resourceLoader.getResource("classpath:images/excel.png");
try (InputStream resourceInputStream = resource.getInputStream()) {
    // 判断是否存在附件，存在则插入表格
    if (!StringUtils.isEmpty(c.getEnclosure()) && !StringUtils.isEmpty(c.getEnclosureName())) {
        HSSFSheet sheet = workbook.createSheet("附件");
        byte[] read = this.fileSystem.read(c.getEnclosure());
        HSSFPatriarch patriarch = sheet.createDrawingPatriarch();
        // anchor主要用于设置图片的属性
        HSSFClientAnchor anchor = new HSSFClientAnchor(0, 0, 0, 0, (short) 0, 0, (short) 3, 10);
        anchor.setAnchorType(ClientAnchor.AnchorType.byId(3));
        // 判断是那种附件类型
        if (c.getEnclosureName().contains(".xlsx") || c.getEnclosureName().contains(".xls")) {
            String attFileName = System.currentTimeMillis() + ".xls";
            // 将附件数据写入Excel
            int attachIndex = workbook.addOlePackage(read, attFileName, attFileName, attFileName);
            // 将文件缩略图写入excel
            int picId = workbook.addPicture(IOUtils.toByteArray(resourceInputStream), HSSFWorkbook.PICTURE_TYPE_JPEG);
            //插入附件
            patriarch.createObjectData(anchor, attachIndex, picId);
        } else {
            //插入图片
            patriarch.createPicture(anchor, workbook.addPicture(read, HSSFWorkbook.PICTURE_TYPE_JPEG));
        }
    }
} catch (IOException e) {
    log.info("导出清单附件出错");
}
